aws_lambda_function

## Example Usage

resource "aws_iam_role" "iam_for_lambda" {
  name = "iam_for_lambda"
  assume_role_policy = <<EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Action": "sts:AssumeRole",
      "Principal": {
        "Service": "lambda.amazonaws.com"
      },
      "Effect": "Allow",
      "Sid": ""
    }
  ]
}
EOF
}

resource "aws_lambda_function" "test_lambda" {
  filename      = "lambda_function_payload.zip"
  function_name = "lambda_function_name"
  role          = aws_iam_role.iam_for_lambda.arn
  handler       = "index.test"
  source_code_hash = filebase64sha256("lambda_function_payload.zip")
  runtime = "nodejs12.x"

  environment {
    variables = {
      foo = "bar"
    }
  }
}

data "archive_file" "example" {
  type        = "zip"
  source_file = "${path.module}/lambda/index.js"
  output_path = "${path.module}/lambda/function.zip"
}

resource "aws_lambda_function" "example" {
  filename      = data.archive_file.example.output_path
  function_name = "example_lambda_function"
  role          = aws_iam_role.example.arn
  handler       = "index.handler"
  code_sha256   = data.archive_file.example.output_base64sha256
  runtime       = "nodejs20.x"

  environment {
    variables = {
      ENVIRONMENT = "production"
      LOG_LEVEL   = "info"
    }
  }
}

## Argument Reference

- function_name (Required): The name of the Lambda function.
- runtime (Required): The runtime environment for the Lambda function.
- handler (Required): The function entrypoint in your code.
- role (Required): The IAM role ARN for the function's execution role.
- filename (Optional): Path to the Lambda function deployment package (ZIP file).
- source_code_hash (Optional): Base64-encoded SHA-256 hash of the deployment package.
- code_sha256 (Optional): Alternative to `source_code_hash` for specifying code integrity.
- image_uri (Optional): The URI of the container image (for container-based functions).
- environment (Optional): Environment variables for the function.
- layers (Optional): List of Lambda Layer ARNs to attach to the function.
- timeout (Optional): Function execution timeout in seconds.
- memory_size (Optional): Amount of memory in MB for the function.
- ephemeral_storage (Optional): Ephemeral storage configuration (up to 10 GB).
- vpc_config (Optional): VPC configuration for the function.
- dead_letter_config (Optional): Dead letter queue configuration.
- tracing_config (Optional): X-Ray tracing configuration.
- kms_key_arn (Optional): KMS key ARN for encryption.
- code_signing_config_arn (Optional): Code signing configuration ARN.
- architectures (Optional): Instruction set architecture (for example `x86_64`, `arm64`).
- tags (Optional): Resource tags.
- description (Optional): Description of the function.
